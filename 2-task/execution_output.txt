-- 2.1. dim_date (Covers 2024–2026 for 24-month rolling balance analysis)
CREATE TABLE dim_date (
    date_key      INTEGER PRIMARY KEY,
    full_date     DATE NOT NULL UNIQUE,
    year          SMALLINT NOT NULL,
    quarter       SMALLINT NOT NULL,
    month_num     SMALLINT NOT NULL,
    year_month    CHAR(7) NOT NULL,                   -- Format: 'YYYY-MM'
    day_of_month  SMALLINT NOT NULL,
    is_weekend    BOOLEAN NOT NULL
);
CREATE TABLE
INSERT INTO dim_date (
    date_key, full_date, year, quarter, month_num, year_month, day_of_month, is_weekend
)
SELECT
    TO_CHAR(d, 'YYYYMMDD')::INTEGER   AS date_key,
    d::DATE                           AS full_date,
    EXTRACT(YEAR FROM d)::SMALLINT    AS year,
    EXTRACT(QUARTER FROM d)::SMALLINT AS quarter,
    EXTRACT(MONTH FROM d)::SMALLINT   AS month_num,
    TO_CHAR(d, 'YYYY-MM')             AS year_month,
    EXTRACT(DAY FROM d)::SMALLINT     AS day_of_month,
    CASE WHEN EXTRACT(ISODOW FROM d) IN (6, 7) THEN TRUE ELSE FALSE END AS is_weekend
FROM generate_series(DATE '2024-01-01', DATE '2026-12-31', INTERVAL '1 day') AS gs(d);

-- 5. ANALYTICAL QUERIES ANSWERING MANAGEMENT QUESTIONS 1–5
-- ============================================================================
-- ----------------------------------------------------------------------------
-- QUESTION 1: Daily net inflow of deposits by region and by customer segment,
--             with yesterday's figure available by 08:00.
-- ----------------------------------------------------------------------------
SELECT
    d.full_date                                                     AS report_date,
    b.region,
    c.segment                                                       AS customer_segment,
    SUM(f.inflow_amount_uzs)                                        AS total_inflow_uzs,
    SUM(f.outflow_amount_uzs)                                       AS total_outflow_uzs,
    SUM(f.inflow_amount_uzs - f.outflow_amount_uzs)                 AS net_inflow_uzs,
    MAX(TO_CHAR(f.batch_loaded_at, 'YYYY-MM-DD HH24:MI'))           AS available_at_local,
    CASE
        WHEN MAX(EXTRACT(HOUR FROM f.batch_loaded_at)) < 8 THEN 'MET (<08:00 SLA)'
        ELSE 'BREACHED'
    END                                                             AS sla_status
FROM fact_deposit_flows f
JOIN dim_date d          ON d.date_key    = f.date_key
JOIN dim_branch b        ON b.branch_key  = f.branch_key
JOIN dim_customer_scd2 c ON c.customer_sk = f.customer_sk
WHERE d.full_date = DATE '2026-11-07'
GROUP BY d.full_date, b.region, c.segment
ORDER BY net_inflow_uzs DESC;

-- ----------------------------------------------------------------------------
-- QUESTION 3: Share of card transactions declined, by merchant category and
--             hour of day, within 5 minutes of the event.
-- ----------------------------------------------------------------------------
SELECT
    DATE_TRUNC('hour', t.event_timestamp)                           AS hour_bucket,
    EXTRACT(HOUR FROM t.event_timestamp)::INTEGER                   AS hour_of_day,
    m.merchant_category,
    COUNT(*)                                                        AS total_transactions,
    COUNT(*) FILTER (WHERE t.tx_status = 'DECLINED')                AS declined_transactions,
    ROUND(
        100.0 * COUNT(*) FILTER (WHERE t.tx_status = 'DECLINED') / NULLIF(COUNT(*), 0),
        2
    )                                                               AS decline_share_pct,
    ROUND(
        AVG(EXTRACT(EPOCH FROM (t.ingested_at - t.event_timestamp))),
        2
    )                                                               AS avg_stream_latency_sec
FROM fact_card_transactions t
JOIN dim_merchant m ON m.merchant_key = t.merchant_key
WHERE t.ingested_at <= t.event_timestamp + INTERVAL '5 minutes'
GROUP BY DATE_TRUNC('hour', t.event_timestamp), EXTRACT(HOUR FROM t.event_timestamp), m.merchant_category
ORDER BY hour_bucket, decline_share_pct DESC;

-- ----------------------------------------------------------------------------
-- QUESTION 4A: Time from loan application to disbursement, by branch and loan product
-- ----------------------------------------------------------------------------
SELECT
    b.branch_code,
    b.branch_name,
    lp.product_category,
    COUNT(*) FILTER (WHERE f.current_stage = '5_DISBURSED')         AS disbursed_count,
    ROUND(AVG(EXTRACT(EPOCH FROM (f.kyc_completed_at - f.applied_at)) / 3600.0), 1)       AS avg_kyc_hours,
    ROUND(AVG(EXTRACT(EPOCH FROM (f.underwriting_at - f.kyc_completed_at)) / 3600.0), 1)  AS avg_underwriting_hours,
    ROUND(AVG(EXTRACT(EPOCH FROM (f.approved_at - f.underwriting_at)) / 3600.0), 1)       AS avg_approval_hours,
    ROUND(AVG(EXTRACT(EPOCH FROM (f.disbursed_at - f.applied_at)) / 3600.0), 1)           AS avg_total_hours_to_disburse,
    lp.target_sla_hours
FROM fact_loan_pipeline f
JOIN dim_branch b        ON b.branch_key       = f.branch_key
JOIN dim_loan_product lp ON lp.loan_product_key = f.loan_product_key
WHERE f.current_stage = '5_DISBURSED'
GROUP BY b.branch_code, b.branch_name, lp.product_category, lp.target_sla_hours
ORDER BY avg_total_hours_to_disburse DESC;

-- QUESTION 4B: Bottleneck analysis — Where pending loan applications stall
-- ----------------------------------------------------------------------------
SELECT
    f.current_stage                                                 AS stalled_at_stage,
    b.branch_code,
    lp.product_category,
    COUNT(*)                                                        AS stalled_applications,
    ROUND(
        AVG(EXTRACT(EPOCH FROM (TIMESTAMPTZ '2026-11-08 07:00:00+05' - COALESCE(f.approved_at, f.underwriting_at, f.kyc_completed_at, f.applied_at))) / 3600.0),
        1
    )                                                               AS hours_stuck_in_current_stage
FROM fact_loan_pipeline f
JOIN dim_branch b        ON b.branch_key       = f.branch_key
JOIN dim_loan_product lp ON lp.loan_product_key = f.loan_product_key
WHERE f.current_stage <> '5_DISBURSED'
GROUP BY f.current_stage, b.branch_code, lp.product_category
ORDER BY hours_stuck_in_current_stage DESC;
